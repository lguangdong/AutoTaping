> **变更日期：2026-11-08（S7）**：新增 `DowntimeRecord` 表（产量页停机分析），
> `IAlarmRepository` 增 `Close` / `QueryActive`，`IOutputRepository` 增 `Query`，
> 新增 `IDowntimeRepository` / `IShiftProductionRepository`。变更原因见各节注释。
>
> **变更日期：2026-11-08（S9）**：`IUserRepository` 增 `GetAll` / `GetById` / `ExistsUserName` /
> `Insert` / `Update` / `Delete`（用户管理：增删改、改密、禁用）；表结构未变（`User` 表已满足需要，
> 登录名不可改，禁用用 `IsEnabled`）。数据保留策略（第 7 节）由 `DailyMaintenanceService` 实现。
>
> 技术选型：**SQLite + Dapper**。单机、零部署、免安装。
> 数据库文件：`data/autotaping.db`（路径来自 `appsettings.json`）

## 2. 建表 SQL

```sql
PRAGMA journal_mode = WAL;   -- 断电安全性 + 并发读

-- ============ 配方 ============
CREATE TABLE IF NOT EXISTS Recipe (
    Id            INTEGER PRIMARY KEY AUTOINCREMENT,
    Name          TEXT    NOT NULL,              -- 配方名，如 "A123-3M双面胶"
    ProductCode   TEXT    NOT NULL DEFAULT '',   -- 产品料号
    TapeSpec      TEXT    NOT NULL DEFAULT '',   -- 胶带规格
    JsonData      TEXT    NOT NULL,              -- 完整参数 JSON（见第 4 节）
    Version       INTEGER NOT NULL DEFAULT 1,
    IsActive      INTEGER NOT NULL DEFAULT 0,    -- 当前激活配方
    IsDeleted     INTEGER NOT NULL DEFAULT 0,    -- 软删除
    CreatedAt     TEXT    NOT NULL,
    CreatedBy     TEXT    NOT NULL DEFAULT '',
    UpdatedAt     TEXT    NOT NULL,
    UpdatedBy     TEXT    NOT NULL DEFAULT ''
);
CREATE UNIQUE INDEX IF NOT EXISTS UX_Recipe_Name ON Recipe(Name) WHERE IsDeleted = 0;

-- ============ 报警 ============
CREATE TABLE IF NOT EXISTS AlarmRecord (
    Id          INTEGER PRIMARY KEY AUTOINCREMENT,
    Code        TEXT    NOT NULL,       -- AL101
    Message     TEXT    NOT NULL,
    Level       INTEGER NOT NULL,       -- 1=提示 2=警告 3=严重
    Source      TEXT    NOT NULL DEFAULT '',
    StartTime   TEXT    NOT NULL,
    AckTime     TEXT,                   -- 确认时间
    AckUser     TEXT,
    EndTime     TEXT,                   -- 恢复时间（条件消失）
    IsAcked     INTEGER NOT NULL DEFAULT 0
);
CREATE INDEX IF NOT EXISTS IX_Alarm_Time ON AlarmRecord(StartTime DESC);
CREATE INDEX IF NOT EXISTS IX_Alarm_Acked ON AlarmRecord(IsAcked, Level);

-- ============ 产出记录（每件一条，追溯核心）============
CREATE TABLE IF NOT EXISTS OutputRecord (
    Id            INTEGER PRIMARY KEY AUTOINCREMENT,
    SequenceNo    INTEGER NOT NULL,      -- 当日流水号
    RecipeId      INTEGER NOT NULL,
    RecipeName    TEXT    NOT NULL,
    Result        TEXT    NOT NULL,      -- 'OK' / 'NG'
    CycleTimeMs   INTEGER NOT NULL,
    DeltaX        REAL    NOT NULL DEFAULT 0,   -- 纠偏量，用于事后分析
    DeltaY        REAL    NOT NULL DEFAULT 0,
    DeltaTheta    REAL    NOT NULL DEFAULT 0,
    MatchScore1   REAL    NOT NULL DEFAULT 0,   -- 模板匹配得分，低于阈值要关注
    MatchScore2   REAL    NOT NULL DEFAULT 0,
    NgReason      TEXT,
    Shift         TEXT    NOT NULL,      -- 'DAY'/'SWING'/'NIGHT'
    CreatedAt     TEXT    NOT NULL
);
CREATE INDEX IF NOT EXISTS IX_Output_Time ON OutputRecord(CreatedAt DESC);
CREATE INDEX IF NOT EXISTS IX_Output_Result ON OutputRecord(Result, CreatedAt DESC);

-- ============ 操作日志 ============
CREATE TABLE IF NOT EXISTS OperationLog (
    Id        INTEGER PRIMARY KEY AUTOINCREMENT,
    UserName  TEXT    NOT NULL,
    Action    TEXT    NOT NULL,   -- '修改参数' / '启动' / '标定' / '登录'
    Target    TEXT    NOT NULL DEFAULT '',
    OldValue  TEXT,
    NewValue  TEXT,
    CreatedAt TEXT    NOT NULL
);
CREATE INDEX IF NOT EXISTS IX_OpLog_Time ON OperationLog(CreatedAt DESC);

-- ============ 标定记录 ============
CREATE TABLE IF NOT EXISTS Calibration (
    Id          INTEGER PRIMARY KEY AUTOINCREMENT,
    MatrixA     REAL NOT NULL, MatrixB REAL NOT NULL, MatrixC REAL NOT NULL,
    MatrixD     REAL NOT NULL, MatrixE REAL NOT NULL, MatrixF REAL NOT NULL,
    RmsError    REAL NOT NULL,
    PointCount  INTEGER NOT NULL,
    ImageFolder TEXT NOT NULL,      -- 标定原图目录
    IsActive    INTEGER NOT NULL DEFAULT 0,
    CreatedAt   TEXT NOT NULL,
    CreatedBy   TEXT NOT NULL DEFAULT ''
);

-- ============ 用户 ============
CREATE TABLE IF NOT EXISTS User (
    Id            INTEGER PRIMARY KEY AUTOINCREMENT,
    UserName      TEXT    NOT NULL UNIQUE,
    PasswordHash  TEXT    NOT NULL,   -- SHA256(salt + password)
    Salt          TEXT    NOT NULL,
    Role          INTEGER NOT NULL,   -- 1=操作员 2=技术员 3=工程师
    IsEnabled     INTEGER NOT NULL DEFAULT 1,
    CreatedAt     TEXT    NOT NULL,
    LastLoginAt   TEXT
);

-- ============ 班次产量 ============
CREATE TABLE IF NOT EXISTS ShiftProduction (
    Id              INTEGER PRIMARY KEY AUTOINCREMENT,
    ShiftDate       TEXT NOT NULL,      -- yyyy-MM-dd
    ShiftName       TEXT NOT NULL,      -- DAY/SWING/NIGHT
    TotalCount      INTEGER NOT NULL DEFAULT 0,
    OkCount         INTEGER NOT NULL DEFAULT 0,
    NgCount         INTEGER NOT NULL DEFAULT 0,
    RunSeconds      INTEGER NOT NULL DEFAULT 0,
    DownSeconds     INTEGER NOT NULL DEFAULT 0,
    LastUpdateAt    TEXT NOT NULL,
    UNIQUE(ShiftDate, ShiftName)
);

-- ============ 停机记录（S7 新增）============
-- 分类口径见 05-核心状态机与流程 第 7.2 节；由协调器在暂停/故障/待料/断电时写入
CREATE TABLE IF NOT EXISTS DowntimeRecord (
    Id              INTEGER PRIMARY KEY AUTOINCREMENT,
    Reason          TEXT    NOT NULL,   -- WAIT_MATERIAL / FAULT / MANUAL_PAUSE / CHANGE_RECIPE / POWER_OFF
    StartTime       TEXT    NOT NULL,
    EndTime         TEXT,               -- 未知（断电补记）为 NULL
    DurationSeconds INTEGER NOT NULL DEFAULT 0
);
CREATE INDEX IF NOT EXISTS IX_Downtime_Time ON DowntimeRecord(StartTime DESC);

-- ============ 系统状态（断电检测）============
CREATE TABLE IF NOT EXISTS SystemState (
    Key        TEXT PRIMARY KEY,
    Value      TEXT NOT NULL,
    UpdatedAt  TEXT NOT NULL
);
-- 关键记录：Key='LastShutdownClean'，正常退出时写 '1'，启动时读取并立即置 '0'
```

## 4. 配方参数结构（`JsonData` 的内容）

参数分四组，JSON 顶层键与 `ParameterGroup` 枚举一一对应：

| 分组键 | 含义 | 说明 |
|---|---|---|
| `process` | 工艺参数 | 速度、高度、压力、保压/真空/吹气延时 |
| `position` | 位置参数 | **头坐标**（龙门 X/Y 站位与 R 角度），不再是旧的「平台坐标」 |
| `vision` | 视觉参数 | 搜索区域、两个大小圆 MARK 的半径、MARK 基准像素、匹配阈值、纠偏上限 |
| `alarm` | 报警阈值 | 真空超时、位置容差 |

```jsonc
{
  "process": {
    "safeHeightMm": 50.0,
    "tapeHeightMm": 2.0,
    "moveSpeedMmPerSec": 100.0,
    "rotateSpeedDegPerSec": 90.0,
    "holdPressureMs": 500,
    "holdPressureMpa": 0.25,
    "vacuumWaitMs": 1500,
    "blowOffMs": 300
  },
  "position": {
    "photoPosX": 150.0,
    "photoPosY": 200.0,
    "tapeBaseX": 220.0,
    "tapeBaseY": 200.0,
    "tapeBaseR": 0.0,
    "pickPosX": 200.0,
    "pickPosY": 80.0,
    "cameraOffsetX": 0.0,
    "cameraOffsetY": 44.0,
    "rotationCenterX": 220.0,
    "rotationCenterY": 200.0
  },
  "vision": {
    // 两个 MARK 共用的搜索区域：标称位置 (400,500)/(880,500) ± 100px。
    // 职责只有「省算力 + 防误检」；模板制作不得改写它（改写会让搜索区退化成模板尺寸 → 1×1）。
    "searchRoi": { "x": 300, "y": 400, "w": 680, "h": 200 },
    // 两个 MARK 是不同直径的实心圆，身份靠直径区分；半径至少差 4px 才能可靠分开
    "mark1RadiusPx": 15.0,
    "mark2RadiusPx": 9.0,
    // 两点间距偏离标称值超过它即判识别失败（防"某个模板命中了干扰圆"）
    "markDistanceTolerancePx": 20.0,
    "template1": "templates/{recipe}/mark1.png",
    "template2": "templates/{recipe}/mark2.png",
    "templateMark1U": 400.0,
    "templateMark1V": 500.0,
    "templateMark2U": 880.0,
    "templateMark2V": 500.0,
    "matchThreshold": 0.7,
    "maxDeltaXMm": 5.0,
    "maxDeltaYMm": 5.0,
    "maxDeltaThetaDeg": 10.0
  },
  // 点贴不再使用「贴附路径」，此处仅为兼容旧配方 JSON 保留示例：
  // 代码里的 RecipeParameters 已删除 TapePath 属性，读旧数据时该节点被 System.Text.Json 忽略。
  "tapePath": {
    "mode": "Line",
    "points": [
      { "x": 0.0,  "y": 0.0,  "z": 2.0 },
      { "x": 30.0, "y": 0.0,  "z": 2.0 }
    ],
    "arcCenter": null,
    "arcRadius": 0.0,
    "arcStartDeg": 0.0,
    "arcEndDeg": 0.0
  },
  "alarm": {
    "vacuumTimeoutMs": 1500,
    "positionToleranceMm": 0.05
  }
}
```

**规则**：
- 所有参数必须有默认值，JSON 缺字段时不得抛异常
- 参数修改后必须校验范围（见 `09-报警与参数清单.md` 参数表）
- `{recipe}` 占位符在加载时替换为配方名
- **旧配方兼容**：产线重构删掉了 `tapePath` 节点、`tapeSpeedMmPerSec` 与 `unloadX` / `unloadY` 三个键。
  `RecipeJson.Options` 未配置 `UnmappedMemberHandling.Disallow`，反序列化时未知键**静默忽略**，
  因此 `Recipe.JsonData` 与历史导出文件里的旧字段**无需迁移**，下次保存时自然写成新结构。

## 5. 仓储接口

```csharp
namespace AutoTaping.Core.Abstractions;

public interface IRecipeRepository
{
    IReadOnlyList<Recipe> GetAll();
    Recipe? GetById(int id);
    Recipe? GetActive();
    int Insert(Recipe recipe);
    void Update(Recipe recipe);
    void SetActive(int id);          // 事务：先全部置 0，再置目标为 1
    void SoftDelete(int id);
    string ExportToFile(int id, string targetPath);    // JSON
    int ImportFromFile(string sourcePath);
}

public interface IAlarmRepository
{
    void Insert(AlarmRecord record);
    void Ack(int id, string user);
    void CloseAllActive();
    void Close(int id, DateTime endTime);                     // S7：条件消失自动恢复（09 文档第三节）
    IReadOnlyList<AlarmRecord> QueryActive();                 // S7：报警页「当前报警」列表
    IReadOnlyList<AlarmRecord> Query(DateTime from, DateTime to, bool onlyUnacked = false);
    int CountUnacked(int minLevel);
    void ExportCsv(DateTime from, DateTime to, string path);
}

public interface IOutputRepository
{
    void Insert(OutputRecord record);
    int CountToday(string result);           // 'OK'/'NG'
    int NextSequenceNo(DateTime date);
    OutputStatistics GetStatistics(DateTime from, DateTime to);
    IReadOnlyList<OutputRecord> Query(DateTime from, DateTime to, string? shift, int maxCount);  // S7：产量页明细
    void ExportCsv(DateTime from, DateTime to, string path);
}

public interface IOperationLogRepository
{
    void Insert(string user, string action, string target, string? oldValue, string? newValue);
    IReadOnlyList<OperationLog> Query(DateTime from, DateTime to);
}

// S7 新增：停机记录与班次汇总（产量页停机分析 / 班次产量）
public interface IDowntimeRepository
{
    int Insert(DowntimeRecord record);
    IReadOnlyList<DowntimeRecord> Query(DateTime from, DateTime to);
}

public interface IShiftProductionRepository
{
    void Upsert(ShiftProduction summary);        // ShiftDate + ShiftName 唯一，可反复重算
    IReadOnlyList<ShiftProduction> Query(DateTime date);
}

// S9 扩展：用户管理（增删改 / 改密 / 禁用），业务校验与审计在 UserManagementService 里
public interface IUserRepository
{
    User? GetByUserName(string userName);        // 登录用，大小写不敏感
    void UpdateLastLogin(int userId, DateTime time);
    IReadOnlyList<User> GetAll();                // 角色倒序 + 用户名升序
    User? GetById(int userId);
    bool ExistsUserName(string userName);        // 新增前的唯一性校验
    int Insert(User user);                       // 返回新主键
    void Update(User user);                      // 只改角色 / 启用状态 / 盐 + 哈希，登录名不可改
    void Delete(int userId);                     // 物理删除；产出与操作日志中的历史记录保留
}
```

## 6. 关键查询

```sql
-- 当班良率
SELECT
    COUNT(*)                                        AS Total,
    SUM(CASE WHEN Result='OK' THEN 1 ELSE 0 END)     AS OkCount,
    AVG(CycleTimeMs)                                AS AvgCycleMs
FROM OutputRecord
WHERE ShiftDate = @date AND Shift = @shift;

-- 当前未确认的严重报警数（用于互锁判断）
SELECT COUNT(*) FROM AlarmRecord
WHERE IsAcked = 0 AND Level = 3 AND EndTime IS NULL;

-- 当日流水号
SELECT IFNULL(MAX(SequenceNo), 0) + 1
FROM OutputRecord
WHERE CreatedAt >= @todayStart;
```

## 7. 数据保留策略

| 表 | 保留 | 处理 |
|---|---|---|
| `OutputRecord` | 90 天 | 每日 02:00 归档到 CSV 后删除 |
| `AlarmRecord` | 180 天 | 同上 |
| `OperationLog` | 365 天 | 同上 |
| `ShiftProduction` | 永久 | 数据量小 |
| `Calibration` | 最近 20 条 | 超出删除最旧（含图片目录） |

**归档任务**：`DailyMaintenanceService`（`PeriodicTimer` + 每天固定时间执行），必须在后台线程，不得阻塞 UI。

## 8. 断电安全

```csharp
// 启动时
var lastClean = _stateRepo.Get("LastShutdownClean");
if (lastClean != "1")
{
    // 上次非正常退出 → 补记一条停机原因
    _output.RecordDowntime("POWER_OFF", estimatedMinutes: null);
    _alarm.Raise("AL503", level: 2, "检测到上次非正常退出");
}
_stateRepo.Set("LastShutdownClean", "0");

// 正常退出时（App.OnExit）
_stateRepo.Set("LastShutdownClean", "1");
```

**SQLite 必须开启 WAL 模式**，且每次写入使用事务，保证断电后数据一致。

## 10. 本文件的约束

1. 不得使用 EF Core / ORM
2. 所有写操作必须包在事务里
3. 密码必须加盐哈希（SHA256 + 随机 16 字节盐），**禁止明文 / 禁止无盐 MD5**
4. 删除配方必须软删除（`IsDeleted = 1`），不得物理删除
5. 所有时间统一存 `TEXT`，格式 `yyyy-MM-dd HH:mm:ss`
6. 导出 CSV 必须是 UTF-8 with BOM
